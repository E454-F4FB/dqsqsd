    local eventDelay = 5000  -- 5 secondes entre les événements fictifs

    -- Génère des événements faux pour brouiller les logs

        Citizen.Wait(eventDelay)

        -- Simule un événement de mouvement aléatoire (pour éviter les checks de vitesse)
        local x, y, z = GetEntityCoords(PlayerPed())
        local offsetX, offsetY = math.random(-5, 5), math.random(-5, 5)
        SetEntityCoordsNoOffset(PlayerPed(), x + offsetX, y + offsetY, z, false, false, false)

        -- Simule une interaction avec un objet (pour éviter les checks de "freeze")
        local handle = GetClosestObjectOfType(x, y, z, 1.0, 288350461, false, false, false)
        if DoesEntityExist(handle) then
            NetworkRequestControlOfEntity(handle)
            Citizen.Wait(100)
            ReleaseControlOfEntity(handle)
        end

        -- Spoof un événement de santé (pour éviter les détections de "health hack")
        TriggerServerEvent("player:updateHealth", 200, 100)
        Citizen.Wait(100)
        TriggerServerEvent("player:updateHealth", 199, 99)  -- Réinitialise pour éviter les logs
    end
end)

-- ===== [7] TELEPORTATION AVANCÉE (Avec direction) =====
function TeleportInDirection(distance)
    local forwardX, forwardY = table.unpack(GetEntityForwardVector(PlayerPed()))
    local x, y, z = table.unpack(GetOffsetFromEntityInWorldCoords(PlayerPed(), 0.0, distance, 0.0))
    TeleportToCoords(x, y, z)
end

-- Bind pour téléporter vers l'avant (LShift)
Citizen.CreateThread(function()
    while true do
        Citizen.Wait(0)
        if IsControlPressed(1, 21) and IsControlJustPressed(1, 21) then  -- LShift
            TeleportInDirection(50.0)  -- 50 mètres devant
        end
    end
end)

-- ===== [8] INVISIBILITÉ OPTIONNELLE (Client-Side) =====
function ToggleInvisibility()
    local ped = PlayerPedId()
    if not IsEntityVisible(ped) then
        SetEntityVisible(ped, true, 0)
        Notify("~r~Invisibilité ~w~: ~b~DÉSACTIVÉE")
    else
        SetEntityVisible(ped, false, 0)
        Notify("~r~Invisibilité ~w~: ~g~ACTIVÉE")
    end
end

-- Bind pour l'invisibilité (F5)
Citizen.CreateThread(function()
    while true do
        Citizen.Wait(0)
        if IsControlJustPressed(1, 165) then  -- F5
            ToggleInvisibility()
        end
    end
end)
